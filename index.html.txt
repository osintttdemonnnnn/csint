let runs=JSON.parse(localStorage.getItem("aegisRuns")||"[]");
let cases=JSON.parse(localStorage.getItem("aegisCases")||"[]");
let favs=new Set(JSON.parse(localStorage.getItem("aegisFavs")||"[]"));
let activeCategory="All", selectedTool=null;

const $=s=>document.querySelector(s);
const grid=$("#toolGrid"), category=$("#category"), search=$("#search");

function esc(v){return String(v).replace(/[&<>"']/g,m=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"}[m]));}

function render(){
  const q=search.value.toLowerCase();
  const visible=tools.filter(t=>
    (activeCategory==="All"||t.cat===activeCategory) &&
    (t.name.toLowerCase().includes(q)||t.cat.toLowerCase().includes(q)||t.desc.toLowerCase().includes(q))
  );
  grid.innerHTML=visible.length?visible.map(t=>`
    <article class="tool-card">
      <div class="tool-head"><div class="icon">${t.icon}</div>
      <button class="fav ${favs.has(t.id)?"on":""}" data-fav="${t.id}">${favs.has(t.id)?"★":"☆"}</button></div>
      <h3>${esc(t.name)}</h3><p>${esc(t.desc)}</p>
      <span class="tag">${esc(t.cat)}</span><span class="tag">Authorized use</span>
      <button class="run" data-run="${t.id}">Open tool ↗</button>
    </article>`).join(""):`<div class="empty">No tools match your search.</div>`;

  $("#toolCount").textContent=tools.length;
  $("#runCount").textContent=runs.length;
  $("#caseCount").textContent=cases.length;
  $("#evidenceCount").textContent=runs.length;
  renderActivity();
}

function renderActivity(){
  $("#activity").innerHTML=runs.length?runs.slice(0,7).map(r=>`
    <div class="activity-item"><b>${esc(r.tool)}</b><small>${esc(r.target)} · ${esc(r.time)}</small></div>
  `).join(""):`<div class="activity-item"><small>No runs yet. Open a tool to begin.</small></div>`;
}

function setCategory(cat){
  activeCategory=cat; category.value=cat;
  document.querySelectorAll("[data-filter]").forEach(b=>b.classList.toggle("active",b.dataset.filter===cat));
  render();
}

[...new Set(tools.map(t=>t.cat))].forEach(c=>{
  const o=document.createElement("option");o.value=c;o.textContent=c;category.appendChild(o);
});
category.insertAdjacentHTML("afterbegin",'<option value="All">All categories</option>');
category.onchange=()=>setCategory(category.value);
search.oninput=render;

document.addEventListener("click",e=>{
  const filter=e.target.closest("[data-filter]");
  if(filter)setCategory(filter.dataset.filter);
  const fav=e.target.closest("[data-fav]");
  if(fav){
    const id=fav.dataset.fav;
    favs.has(id)?favs.delete(id):favs.add(id);
    localStorage.setItem("aegisFavs",JSON.stringify([...favs]));render();
  }
  const run=e.target.closest("[data-run]");
  if(run)openTool(run.dataset.run);
});

function openTool(id){
  selectedTool=tools.find(t=>t.id===id);
  $("#modalCategory").textContent=selectedTool.cat;
  $("#modalTitle").textContent=selectedTool.name;
  $("#modalDescription").textContent=selectedTool.desc;
  $("#targetInput").placeholder=selectedTool.placeholder;
  $("#targetInput").value="";
  $("#notes").value="";
  $("#result").style.display="none";
  $("#fileInput").style.display=["image-meta","image-hash","ocr","screenshot","pdf-meta","doc-text"].includes(id)?"block":"none";
  $("#targetInput").style.display=$("#fileInput").style.display==="block"?"none":"block";
  $("#modal").classList.add("open");
}

function closeModal(){$("#modal").classList.remove("open");selectedTool=null}
$("#closeModal").onclick=closeModal;$("#cancelModal").onclick=closeModal;
$("#modal").onclick=e=>{if(e.target.id==="modal")closeModal()};

$("#runTool").onclick=()=>{
  if(!selectedTool)return;
  const file=$("#fileInput").files[0];
  const target=file?file.name:$("#targetInput").value.trim();
  if(!target){
    $("#result").style.display="block";
    $("#result").innerHTML="<strong>Input required.</strong><br>Enter a target or choose a file before running this tool.";
    return;
  }
  const now=new Date().toLocaleString();
  runs.unshift({tool:selectedTool.name,target,time:now,notes:$("#notes").value.trim()});
  localStorage.setItem("aegisRuns",JSON.stringify(runs.slice(0,100)));
  $("#result").style.display="block";
  $("#result").innerHTML=`
    <div class="demo">DEMO RESULT — no external provider was contacted</div>
    <strong>${esc(selectedTool.name)} accepted the input.</strong><br>
    Target: ${esc(target)}<br>
    Scope: public or authorized sources only.<br>
    Next step: connect this tool to a server-side adapter with approved data providers.
  `;
  render();
};

$("#newCase").onclick=()=>{
  const name=prompt("Case name:");
  if(name&&name.trim()){
    cases.unshift({name:name.trim(),created:new Date().toLocaleString()});
    localStorage.setItem("aegisCases",JSON.stringify(cases));
    render();
  }
};

$("#clearLog").onclick=()=>{
  if(confirm("Reset local demo history, cases, and favorites?")){
    runs=[];cases=[];favs.clear();
    localStorage.removeItem("aegisRuns");
    localStorage.removeItem("aegisCases");
    localStorage.removeItem("aegisFavs");
    render();
  }
};

$("#exportLog").onclick=()=>{
  const blob=new Blob([JSON.stringify({exportedAt:new Date().toISOString(),runs,cases},null,2)],{type:"application/json"});
  const a=document.createElement("a");
  a.href=URL.createObjectURL(blob);a.download="aegis-csint-log.json";a.click();
  URL.revokeObjectURL(a.href);
};
